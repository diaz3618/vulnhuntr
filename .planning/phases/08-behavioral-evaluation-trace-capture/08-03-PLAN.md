def _make_subprocess_result(stdout: str, returncode: int = 0, stderr: str = "") -> MagicMock:
    """Build a MagicMock that looks like subprocess.CompletedProcess."""
    m = MagicMock()
    m.stdout = stdout
    m.returncode = returncode
    m.stderr = stderr
    return m

class TestProbeToSendTrace:
    """probe_result and response_validated events are emitted with correct content."""

    def _make_version_result(self) -> MagicMock:
        return _make_subprocess_result("2.1.126 (Claude Code)")

    def _make_send_result(self) -> MagicMock:
        payload = {
            "result": "no vulnerability found",
            "usage": {"input_tokens": 10, "output_tokens": 5},

            "session_id": "sess-abc",

class TestCLIPolicyBoundaries:
    """Equivalence-partition and boundary tests for CLIPolicy field validation.

    Partitions:
      session_mode: valid={"stateless","continue","resume"}, invalid={"unknown"}
      mcp_mode: valid={"none","vulnhuntr","provider","both"}, invalid={"auto"}
      tool_mode: valid={"none","read-only","full"}, invalid={""}
      timeout: boundary=0,1,300(default),sys.maxsize; invalid=-1
      max_turns: boundary=1,10(default),100; invalid=0
      overrides: empty={}, valid nested dict, nested dict with unrecognized key
    """

    import sys

    @pytest.mark.parametrize(
        "session_mode,expect_error",
        [
            pytest.param("stateless", False, id="valid-stateless"),
            pytest.param("continue", False, id="valid-continue"),
            pytest.param("resume", False, id="valid-resume"),
            pytest.param("unknown", True, id="invalid-unknown"),
        ],
    )
    def test_session_mode(self, session_mode: str, expect_error: bool) -> None:
        if expect_error:
            with pytest.raises(ValueError, match="session_mode"):
                CLIPolicy(session_mode=session_mode)
        else:
            p = CLIPolicy(session_mode=session_mode)
            assert p.session_mode == session_mode

    @pytest.mark.parametrize(
        "mcp_mode,expect_error",
        [
            pytest.param("none", False, id="valid-none"),
