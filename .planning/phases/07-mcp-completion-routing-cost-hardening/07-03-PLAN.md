### Task 1: MCP integration tests in `tests/test_mcp_analysis.py`

**File:** `tests/test_mcp_analysis.py`
**Type:** modify

Add a new test class `TestMCPFullLoop` at the end of the file. The class tests
the complete MCP tool call → result → next-iteration flow by exercising
`VulnerabilityAnalyzer._secondary_analysis()` with a mock `mcp_helper`.

**Pattern to follow:** The existing tests in the file use `MCPAnalysisHelper` with
direct `_tools`/`_initialized` injection (not `asyncio.run`). For the full-loop
tests, use `unittest.mock.MagicMock` for the helper because `_secondary_analysis()`
uses `run_async()` (which calls `asyncio.run()`) — but the `execute_tool_calls`
coroutine itself needs to be an `AsyncMock`.

```python
class TestMCPFullLoop:
    """Integration tests for the VulnerabilityAnalyzer MCP tool call loop.

    These tests verify D-05 (tool call → result → next-iteration injection)
    and D-07/D-08 (timeout → MCPToolCallResult(success=False), no raise).
    """

    def _make_mock_helper(self, tool_results=None, active=True):
        """Build a MagicMock MCPAnalysisHelper with AsyncMock execute_tool_calls."""
        from unittest.mock import AsyncMock, MagicMock
        helper = MagicMock()
        helper.is_active = active
        helper.execute_tool_calls = AsyncMock(return_value=tool_results or [])
        # format_results_for_prompt is sync
        if tool_results:
            xml = "<mcp_tool_results>\n  <tool_result>mock output</tool_result>\n</mcp_tool_results>"
        else:
            xml = ""
        helper.format_results_for_prompt.return_value = xml
        return helper

    def _make_analyzer_with_mock_llm(self, mcp_helper=None, max_iterations=2):
        """Build a VulnerabilityAnalyzer with a mock LLM that returns canned Responses."""
        from pathlib import Path
        from unittest.mock import MagicMock
        from vulnhuntr.core.analysis import AnalysisConfig, VulnerabilityAnalyzer
        from vulnhuntr.core.models import MCPToolCallRequest, Response

        # First call returns a response with a tool call request
        first_response = Response(
            scratchpad="checking",
            analysis="found something suspicious",
            confidence_score=5,
            mcp_tool_calls=[
                MCPToolCallRequest(server="scanner", tool="scan", arguments={"path": "app.py"})
            ],
            context_code=[],
        )
        # Second call returns final response with no tool calls and no context requests
        second_response = Response(
            scratchpad="confirmed",
            analysis="SQL injection confirmed via MCP scan result",
            confidence_score=8,
            mcp_tool_calls=[],
            context_code=[],
        )

        mock_llm = MagicMock()
        mock_llm.chat.side_effect = [first_response, second_response]
        mock_llm.set_context = MagicMock()

        mock_extractor = MagicMock()
        mock_extractor.extract.return_value = None

        config = AnalysisConfig(max_iterations=max_iterations)
        analyzer = VulnerabilityAnalyzer(
            llm=mock_llm,
            code_extractor=mock_extractor,
            config=config,
            mcp_helper=mcp_helper,
        )
        return analyzer, mock_llm

    def test_tool_call_result_injected_into_next_iteration(self):
        """When LLM returns mcp_tool_calls, results appear in the next iteration's prompt."""
        from pathlib import Path
        from vulnhuntr.core.models import MCPToolCallResult
        from vulnhuntr.core.models import VulnType

        result = MCPToolCallResult(server="scanner", tool="scan", success=True, output="vuln found")
        helper = self._make_mock_helper(tool_results=[result])
        analyzer, mock_llm = self._make_analyzer_with_mock_llm(mcp_helper=helper)

        _, _ = analyzer._secondary_analysis(
            file_path=Path("app.py"),
            content="x = input()",
            vuln_type=VulnType.SQLI,
            all_files=[Path("app.py")],
            vuln_data={},
            analysis_approach="",
            guidelines="",
        )

        # execute_tool_calls must have been called once (for the first response)
        helper.execute_tool_calls.assert_called_once()

        # The second LLM call's prompt must contain the MCP results XML
        second_call_prompt = mock_llm.chat.call_args_list[1][0][0]
        assert "<mcp_tool_results>" in second_call_prompt, (
            "MCP results XML not found in second iteration prompt"
        )

    def test_no_tool_calls_skips_mcp_entirely(self):
        """When LLM response has no mcp_tool_calls, execute_tool_calls is never called."""
        from pathlib import Path
        from vulnhuntr.core.models import VulnType

        helper = self._make_mock_helper()
        # Override: both responses have no tool calls
        from unittest.mock import MagicMock
        from vulnhuntr.core.analysis import AnalysisConfig, VulnerabilityAnalyzer
        from vulnhuntr.core.models import Response

        resp = Response(
            scratchpad="s", analysis="a", confidence_score=5,
            mcp_tool_calls=[], context_code=[],
        )
        mock_llm = MagicMock()
        mock_llm.chat.return_value = resp
        mock_llm.set_context = MagicMock()
        mock_extractor = MagicMock()
        mock_extractor.extract.return_value = None

        analyzer = VulnerabilityAnalyzer(
            llm=mock_llm,
            code_extractor=mock_extractor,
            config=AnalysisConfig(max_iterations=1),
            mcp_helper=helper,
        )
        analyzer._secondary_analysis(
            file_path=Path("app.py"), content="x=1",
            vuln_type=VulnType.SQLI, all_files=[Path("app.py")],
            vuln_data={}, analysis_approach="", guidelines="",
        )
        helper.execute_tool_calls.assert_not_called()

    def test_none_mcp_helper_runs_without_mcp(self):
        """When mcp_helper is None, secondary analysis runs identically to before."""
        from pathlib import Path
        from vulnhuntr.core.models import VulnType

        analyzer, mock_llm = self._make_analyzer_with_mock_llm(mcp_helper=None)
        report, _ = analyzer._secondary_analysis(
            file_path=Path("app.py"), content="x=input()",
            vuln_type=VulnType.SQLI, all_files=[Path("app.py")],
            vuln_data={}, analysis_approach="", guidelines="",
        )
        assert report is not None

    def test_timeout_produces_failure_result_not_exception(self):
        """asyncio.TimeoutError inside execute_tool_calls → MCPToolCallResult(success=False)."""
        import asyncio
        from unittest.mock import AsyncMock, MagicMock, patch
        from vulnhuntr.mcp.analysis import MCPAnalysisHelper
        from vulnhuntr.mcp.config import MCPAnalysisPolicy, MCPSettings

        settings = MCPSettings()
        helper = MCPAnalysisHelper(settings)
        helper._initialized = True
        helper._tools = [ToolDescriptor("s", "t", "desc")]

        # Patch the client to raise TimeoutError
        mock_client = MagicMock()
        mock_client.call_tool = AsyncMock(side_effect=asyncio.TimeoutError())
        helper._client = mock_client

        from vulnhuntr.core.models import MCPToolCallRequest
        from vulnhuntr.mcp.analysis import run_async

        requests = [MCPToolCallRequest(server="s", tool="t", arguments={})]
        results = run_async(helper.execute_tool_calls(requests))

        assert len(results) == 1
        assert results[0].success is False
        assert "imeout" in results[0].error or "timeout" in results[0].error.lower()

    def test_inactive_helper_skips_tool_dispatch(self):
        """When is_active is False, tool calls in the Response are not dispatched."""
        from pathlib import Path
        from vulnhuntr.core.models import VulnType

        helper = self._make_mock_helper(active=False)
```

## Threat Model

**Security considerations:**

All test mocks use `MagicMock` and `AsyncMock` — no real API keys, CLI binaries,
or network calls are made. The `asyncio.TimeoutError` test patches at the
`_client.call_tool` level, which is well below any network boundary.

No test file writes to disk except via `tmp_path` pytest fixtures (which are
cleaned up automatically).

The `test_cli_provider_sub_spec_logs_warning` test verifies that
`parse_fallback_spec("claude-code:opus")` succeeds and logs a warning rather
than failing — this is the correct security-transparent behavior (the sub-spec
is ignored, not silently used as a model name that could misroute API calls).
