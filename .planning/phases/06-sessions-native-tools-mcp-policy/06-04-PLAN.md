Session mode flag mapping (Claude Code):
  "stateless" → --no-session-persistence
  "continue"  → --continue
  "resume" with session_id → --resume <session_id>
  "resume" without session_id → CLIRuntimeError

Session mode (Gemini/Codex/Qwen Code):
  non-stateless → structured log warning, no exception, proceeds stateless

tool_mode → cmd flag mapping:
  Claude Code:  "none" → --tools "" + --permission-mode bypassPermissions
                "read-only" → --permission-mode plan
                "full" → --permission-mode bypassPermissions (no --tools "")
  Gemini CLI:   "none"/"read-only" → --approval-mode plan
                "full" → --approval-mode yolo
  Codex:        "none"/"read-only" → --sandbox read-only
                "full" → --sandbox workspace-write
  Qwen Code:    "none"/"read-only" → --approval-mode plan
                "full" → --yolo

sandbox_mode (Codex only):
  "workspace-write" → --sandbox workspace-write (overrides tool_mode)
  "none"/"read-only" → --sandbox read-only (unless tool_mode="full")

MCP config (ClaudeCodeLLM only):
  mcp_mode="vulnhuntr"/"both" → writes {"mcpServers": {}} stub, adds --mcp-config
  mcp_mode="none"/"provider" → no --mcp-config flag

class TestClaudeCodeSessionMode:
    """session_mode wire-up in ClaudeCodeLLM.send_message() (D-01, D-02, D-03, D-04)."""

    def _fake_result(self):
        fake = MagicMock()
        fake.stdout = json.dumps(
            {"result": "ok", "usage": {}, "modelUsage": {}, "session_id": "sess-abc"}
        )
        fake.returncode = 0
        return fake

    @pytest.mark.parametrize("mode,expected_flag", [
        ("stateless", "--no-session-persistence"),
        ("continue", "--continue"),
    ])
    def test_session_mode_flag_injected(self, mode, expected_flag):
        from vulnhuntr.cli_providers.claude_code import ClaudeCodeLLM
        from vulnhuntr.config import CLIPolicy
        policy = CLIPolicy(session_mode=mode, mcp_mode="none", tool_mode="none")
        llm = ClaudeCodeLLM(policy=policy)
        with patch("subprocess.run", return_value=self._fake_result()) as m:
            llm.send_message("test", 1024)
        cmd = m.call_args[0][0]
        assert expected_flag in cmd

    def test_resume_with_session_id_injects_flag(self):
        from vulnhuntr.cli_providers.claude_code import ClaudeCodeLLM
        from vulnhuntr.config import CLIPolicy
        policy = CLIPolicy(session_mode="resume", mcp_mode="none")
        llm = ClaudeCodeLLM(policy=policy)
        llm.session_id = "sess-xyz"
        with patch("subprocess.run", return_value=self._fake_result()) as m:
            llm.send_message("test", 1024)
        cmd = m.call_args[0][0]
        assert "--resume" in cmd
        idx = cmd.index("--resume")
        assert cmd[idx + 1] == "sess-xyz"

    def test_resume_without_session_id_raises(self):
        from vulnhuntr.cli_providers.base import CLIRuntimeError
        from vulnhuntr.cli_providers.claude_code import ClaudeCodeLLM
        from vulnhuntr.config import CLIPolicy
        policy = CLIPolicy(session_mode="resume", mcp_mode="none")
        llm = ClaudeCodeLLM(policy=policy)
        with pytest.raises(CLIRuntimeError, match="session_id"):
            llm.send_message("test", 1024)

    def test_session_id_extracted_from_response(self):
        from vulnhuntr.cli_providers.claude_code import ClaudeCodeLLM
        from vulnhuntr.config import CLIPolicy
        policy = CLIPolicy(session_mode="stateless", mcp_mode="none")
        llm = ClaudeCodeLLM(policy=policy)
        with patch("subprocess.run", return_value=self._fake_result()):
            llm.send_message("test", 1024)
        assert llm.session_id == "sess-abc"

    def test_no_session_persistence_only_in_stateless(self):
        """--no-session-persistence must NOT appear for continue mode."""
        from vulnhuntr.cli_providers.claude_code import ClaudeCodeLLM
        from vulnhuntr.config import CLIPolicy
        policy = CLIPolicy(session_mode="continue", mcp_mode="none")
        llm = ClaudeCodeLLM(policy=policy)
        with patch("subprocess.run", return_value=self._fake_result()) as m:
            llm.send_message("test", 1024)
        cmd = m.call_args[0][0]
        assert "--no-session-persistence" not in cmd

class TestCheckpointSessionMetadata:
    """CheckpointData.session_metadata round-trip (D-07)."""

    def test_session_metadata_default_none(self):
        from vulnhuntr.checkpoint import CheckpointData
        cd = CheckpointData()
        assert cd.session_metadata is None

    def test_session_metadata_serializes(self):
        from vulnhuntr.checkpoint import CheckpointData
        meta = {
            "provider": "ClaudeCodeLLM",
            "session_id": "sess-abc",
            "started_at": "2026-05-04T00:00:00",
            "workdir": "/tmp",
            "binary_version": "2.1.0",
        }
        cd = CheckpointData(session_metadata=meta)
        d = cd.to_dict()
        assert "session_metadata" in d
        assert d["session_metadata"] == meta

    def test_session_metadata_deserializes(self):
        from vulnhuntr.checkpoint import CheckpointData
        meta = {"provider": "ClaudeCodeLLM", "session_id": "sess-abc"}
        d = {"session_metadata": meta}
        cd = CheckpointData.from_dict(d)
        assert cd.session_metadata == meta

    def test_session_metadata_missing_key_returns_none(self):
        from vulnhuntr.checkpoint import CheckpointData
        cd = CheckpointData.from_dict({})
        assert cd.session_metadata is None

    def test_session_metadata_none_serializes_as_none(self):
        from vulnhuntr.checkpoint import CheckpointData
        cd = CheckpointData(session_metadata=None)
        d = cd.to_dict()
        assert d["session_metadata"] is None
